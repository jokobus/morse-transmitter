	% Slide 6: Lighthouse Wave Effect
	\begin{frame}[fragile,shrink=10]
		\frametitle{Lighthouse Wave Effect}
		\begin{itemize}
			\item \textbf{Concept:} Light sweeps across LED strip instead of all-on
			\item \textbf{Implementation:}
			\begin{itemize}
				\item Divide dot/dash duration by number of LEDs (8)
				\item Illuminate each LED sequentially
				\item Create smooth sweeping motion
			\end{itemize}
			\item \textbf{Example:} 500 ms dot $\div$ 8 LEDs = 62.5 ms per LED
			\item \textbf{Wave width:} Configurable (5 LEDs default)
			\item \textbf{Boring:} Standard blinking is possible as well, by setting parameter $WAVE\_WIDTH = 0$
		\end{itemize}
		
		\vspace{0.5cm}
		\begin{center}
			\texttt{LED: [1][2][3][4][5][6][7][8]}\\
			\texttt{Time: →→→→→→→→}
		\end{center}
	\end{frame}
	
	% Slide 7: Timer1 Configuration
	\begin{frame}[fragile,shrink=10]
		\frametitle{Hardware Timer1 Setup}
		\begin{itemize}
			\item \textbf{Mode:} CTC (Clear Timer on Compare Match)
			\item \textbf{Prescaler:} 1024
			\item \textbf{Frequency:} 16 MHz / 1024 = 15625 Hz
			\item \textbf{Resolution:} 64 $\mu$s per tick
			\item[] \lstinputlisting[linerange={262-284}, firstnumber=262]{../morse.ino}    
			\item \textbf{Calculation:} ticks = (duration\_ms × 15625) / 1000
		\end{itemize}
		
	\end{frame}

	% Slide 9: ISR Details
	\begin{frame}[fragile,shrink=10]
		\frametitle{ISR Execution Details}
		\textbf{Responsibilities:}
		\begin{enumerate}
			\item Update LED colors and positions
			\item Handle wave effect logic with boundary checking
			\item Transmit data to WS2812B strip (\texttt{show()})
			\item Calculate next timer interval
			\item Advance to next step (with wraparound)
		\end{enumerate}
		
		\vspace{0.5cm}
		\textbf{Timing Considerations:}
		\begin{itemize}
			\item WS2812B update: $\sim$ 240 $\mu s$ for 8 LEDs
			\item Shortest Morse interval: 500 ms
			\item Overhead: 0.05\% (negligible)
		\end{itemize}
	\end{frame}
